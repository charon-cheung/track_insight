cmake_minimum_required(VERSION 3.16)
project(track_insight VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rosbag2_cpp REQUIRED)
find_package(rosbag2_storage REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(action_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(LibArchive REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(yaml-cpp REQUIRED)

add_library(track_insight_core
  src/core/time.cpp src/core/log_parser.cpp src/core/correlation.cpp src/core/tracking_matcher.cpp
  src/core/velocity_aligner.cpp src/core/evaluator.cpp src/core/terminal.cpp src/core/anomalies.cpp src/config.cpp)
target_include_directories(track_insight_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(track_insight_core PUBLIC yaml-cpp)
add_library(track_insight_ros_io
  src/io/input_package.cpp src/io/bag_reader.cpp src/io/tf_reconstructor.cpp)
target_link_libraries(track_insight_ros_io PUBLIC track_insight_core LibArchive::LibArchive OpenSSL::Crypto)
ament_target_dependencies(track_insight_ros_io PUBLIC
  rclcpp rosbag2_cpp rosbag2_storage tf2 tf2_msgs nav_msgs geometry_msgs action_msgs)
  add_library(track_insight_report src/report/serialization.cpp src/report/figures.cpp src/report/report.cpp src/report/overview.cpp src/report/path_filter.cpp src/report/parameter_snapshot.cpp src/report/archive.cpp)
  target_link_libraries(track_insight_report PUBLIC track_insight_core nlohmann_json::nlohmann_json)
  add_executable(track_insight src/main.cpp)
  target_link_libraries(track_insight PRIVATE track_insight_ros_io track_insight_report)
  ament_target_dependencies(track_insight PUBLIC ament_index_cpp)
  install(TARGETS track_insight DESTINATION lib/${PROJECT_NAME})
  install(TARGETS track_insight_report DESTINATION lib)
install(TARGETS track_insight_core track_insight_ros_io DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config resources DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_pipeline test/test_pipeline.cpp TIMEOUT 120)
  target_link_libraries(test_pipeline track_insight_ros_io track_insight_report)
  target_compile_definitions(test_pipeline PRIVATE TRACK_INSIGHT_RESOURCES="${CMAKE_CURRENT_SOURCE_DIR}/resources")
  ament_add_gtest(test_archive test/test_archive.cpp TIMEOUT 120)
  target_link_libraries(test_archive track_insight_report)
  target_compile_definitions(test_archive PRIVATE TRACK_INSIGHT_RESOURCES="${CMAKE_CURRENT_SOURCE_DIR}/resources")
  ament_add_gtest(test_parameters test/test_parameters.cpp)
  target_link_libraries(test_parameters track_insight_report)
  ament_add_gtest(test_overview test/test_overview.cpp)
  target_link_libraries(test_overview track_insight_report)
  ament_add_gtest(test_core test/test_core.cpp)
  target_link_libraries(test_core track_insight_core)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/test/test_input.cpp")
    ament_add_gtest(test_input test/test_input.cpp)
    target_include_directories(test_input PRIVATE src/io)
    target_link_libraries(test_input track_insight_ros_io)
  endif()
endif()
ament_package()
